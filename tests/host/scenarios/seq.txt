# DREADNOUGHT host scenario: the 8-track sequencer from the panel. tests/run_host.sh runs it.
# White keys (steps 1..16): 0 2 4 6 7 9 11 12 14 16 18 19 21 23 24 26.
# Black keys: 1 3 5 8 10 13 15 17 = mute tracks 1..8; 20 = hold to clear; 22 / 25 = steps in view - / +.
# First boot: the demo reese line on track 1
expect on.1 7
expect playing 0
wav seq-demo.wav
tap PLAY
expect playing 1
wait 2790
# two bars at 172 BPM: 192 clock pulses (24 a quarter), give or take the moment PLAY went down
expect clocks >185
expect clocks <200
expect peak >6000
tap SEQ
expect page 5
wait 300
shot seq-01-playing.png
wavstop
# a tap turns a step on, another tap turns it off
tapkey 7 80
expect on.1 8
tapkey 7 80
expect on.1 7
# hold step 1, KNOB 1 up three semitones, KNOB 4 right: slide; letting go keeps it on
key 0 down
turn K1 3
turn K4 1
shot seq-02-step-held.png
key 0 up
expect nt.1.1 56
expect st.1.1 3
# mute track 1 and back
tapkey 1 50
expect mute.1 1
shot seq-03-muted.png
tapkey 1 50
expect mute.1 0
# track 4 (MIDI channel 4): a step, a shorter length, its notes go out on MIDI
turn K1 3
expect trk 4
tapkey 0 80
expect on.4 1
turn K2 -4
expect len.4 4
wait 1000
expect midinotes >1
# swing
turn K3 6
expect swing 60
tap PLAY
expect playing 0
# step record into track 1: clear it first (black key 9 held 1 s)
turn K1 -3
expect trk 1
key 20 down
wait 1200
key 20 up
expect on.1 0
tap SEL
tap REC
expect rec 1
tapkey 0 100
tapkey 3 100
key 0 down
key 4 down
wait 50
key 4 up
key 0 up
tap REC
expect on.1 4
expect nt.1.2 56
expect st.1.3 1
expect st.1.4 3
tap SEQ
shot seq-04-recorded.png
# SAVE keeps the pattern over a restart
expect dirty 1
tap SAVE
reboot
expect on.1 4
expect nt.1.4 57
expect swing 60
expect len.4 4
